/* [page-boot] A read this page already received inside its HTML (Page_Boot,
   injected by doGet) is answered here once, locally. Every other call — and
   every later one — goes on exactly as before, through API._callInner; the
   runtime only watches the declared ones to time the page. */
API.call = function (action, payload, sessionToken, opts) {
  if (window.PageBoot && PageBoot.answer) {
    var booted = PageBoot.answer(action, payload);
    if (booted) return booted;
  }
  var p = API._callInner(action, payload, sessionToken, opts);
  if (window.PageBoot && PageBoot.observe) PageBoot.observe(action, payload, p);
  return p;
};
