// OWNER RUNS in the Apps Script editor (scratch file, then delete it).
// Replaces the real data in erp_test_products and erp_test_customer_vendor with
// generic test data (a public catalogue of common industrial / cleaning
// chemicals, and invented company names).
//
// 1. Run etReplaceMasterDataDryRun  -> logs what would change, writes nothing.
// 2. Run etReplaceMasterDataWrite   -> writes it.
//
// What it keeps, so nothing that points at these rows breaks:
//   - every row's id, and every row's position
//   - products: category, unit, carton, concentration, sales_tax, asset_code,
//     GPC, cost_allocation_percentage, user, audit columns
//   - parties: customer_direction, type, country, region, user, audit columns
// What it replaces:
//   - products: name_ar, name_en
//   - parties:  name, name_en, registration_number, tax_id, telephone, address
//   - erp_test_sales_invoices / erp_test_sales_offer: tax_id, which is a copy of
//     the client's tax_id, is re-copied from the new party row (matched on client_id)
// It bumps `version` on every changed row, so a form still open on an old copy
// is refused on save instead of writing the real data back.
// The output is the same on every run (seeded), so running it twice is harmless.
// It does NOT touch ERP_Record_History: older history entries still hold the
// old values.

var ET_RMD_SS_ID = '1rdnnP3rMZTnoyyfG5V3X6w62AXatgIisZljkg0izJzE';

function etReplaceMasterDataDryRun() { return etReplaceMasterData_(false); }
function etReplaceMasterDataWrite() { return etReplaceMasterData_(true); }

function etReplaceMasterData_(write) {
  var ss = SpreadsheetApp.openById(ET_RMD_SS_ID);
  var report = { write: write };

  // ---- products ----
  var prod = etRmdRead_(ss, 'erp_test_products');
  var prodNew = etRmdProducts_(prod.rows.length);
  report.products = etRmdApply_(prod, prodNew, ['name_ar', 'name_en'], [], write);

  // ---- customers / vendors ----
  var party = etRmdRead_(ss, 'erp_test_customer_vendor');
  var regionCol = party.headers.indexOf('region');
  var partyNew = etRmdParties_(party.rows.map(function (r) { return regionCol === -1 ? '' : String(r.values[regionCol] || '').trim(); }));
  var textCols = ['registration_number', 'tax_id', 'telephone'];
  report.parties = etRmdApply_(party, partyNew, ['name', 'name_en', 'registration_number', 'tax_id', 'telephone', 'address'], textCols, write);

  // ---- tax_id copies on invoice / offer headers ----
  var taxById = {};
  var idCol = party.headers.indexOf('id');
  party.rows.forEach(function (r, i) { taxById[String(r.values[idCol]).trim()] = partyNew[i].tax_id; });
  report.sales_invoices_tax_id = etRmdSyncTax_(ss, 'erp_test_sales_invoices', taxById, write);
  report.sales_offer_tax_id = etRmdSyncTax_(ss, 'erp_test_sales_offer', taxById, write);

  if (write) {
    SpreadsheetApp.flush();
    report.caches = etRmdBustCaches_();
  }
  console.log(JSON.stringify(report, null, 1));
  return report;
}

// Header row + every data row that has an id. Rows without an id are left alone.
function etRmdRead_(ss, name) {
  var sh = ss.getSheetByName(name);
  if (!sh) throw new Error('missing tab: ' + name);
  var lr = sh.getLastRow(), lc = sh.getLastColumn();
  var all = lr ? sh.getRange(1, 1, lr, lc).getValues() : [[]];
  var headers = all[0].map(function (h) { return String(h).trim(); });
  var idCol = headers.indexOf('id');
  if (idCol === -1) throw new Error(name + ': no id column');
  var rows = [];
  for (var i = 1; i < all.length; i++) {
    if (String(all[i][idCol]).trim() === '') continue;
    rows.push({ rowNum: i + 1, values: all[i] });
  }
  return { name: name, sheet: sh, headers: headers, rows: rows };
}

// Writes each target column as one contiguous block from row 2 to the last
// data row, carrying the existing value through for rows without an id.
function etRmdApply_(tab, newRecs, cols, textCols, write) {
  var out = { tab: tab.name, rows: tab.rows.length, columns: [], sample: [] };
  if (!tab.rows.length) return out;
  var first = 2, last = tab.rows[tab.rows.length - 1].rowNum;
  var byRow = {};
  tab.rows.forEach(function (r, i) { byRow[r.rowNum] = newRecs[i]; });
  var present = cols.filter(function (c) { return tab.headers.indexOf(c) !== -1; });
  out.columns = present;
  out.missing_columns = cols.filter(function (c) { return tab.headers.indexOf(c) === -1; });

  tab.rows.slice(0, 5).forEach(function (r, i) {
    var s = { id: r.values[tab.headers.indexOf('id')] };
    present.forEach(function (c) { s[c] = String(r.values[tab.headers.indexOf(c)]) + '  ->  ' + newRecs[i][c]; });
    out.sample.push(s);
  });
  if (!write) return out;

  var current = tab.sheet.getRange(first, 1, last - first + 1, tab.headers.length).getValues();
  present.forEach(function (c) {
    var col = tab.headers.indexOf(c);
    var block = current.map(function (row, k) {
      var rec = byRow[first + k];
      return [rec ? rec[c] : row[col]];
    });
    var range = tab.sheet.getRange(first, col + 1, block.length, 1);
    if (textCols.indexOf(c) !== -1) range.setNumberFormat('@');
    range.setValues(block);
  });

  var vCol = tab.headers.indexOf('version');
  if (vCol !== -1) {
    var vBlock = current.map(function (row, k) {
      if (!byRow[first + k]) return [row[vCol]];
      return [(Number(row[vCol]) || 0) + 1];
    });
    tab.sheet.getRange(first, vCol + 1, vBlock.length, 1).setValues(vBlock);
  }
  return out;
}

function etRmdSyncTax_(ss, name, taxById, write) {
  var sh = ss.getSheetByName(name);
  if (!sh) return { tab: name, skipped: 'missing tab' };
  var lr = sh.getLastRow(), lc = sh.getLastColumn();
  if (lr < 2) return { tab: name, changed: 0 };
  var headers = sh.getRange(1, 1, 1, lc).getValues()[0].map(function (h) { return String(h).trim(); });
  var cCol = headers.indexOf('client_id'); if (cCol === -1) cCol = headers.indexOf('اسم العميل');
  var tCol = headers.indexOf('tax_id'); if (tCol === -1) tCol = headers.indexOf('رقم التسجيل الضريبي للعميل');
  if (cCol === -1 || tCol === -1) return { tab: name, skipped: 'no client_id / tax_id column' };
  var clients = sh.getRange(2, cCol + 1, lr - 1, 1).getValues();
  var taxes = sh.getRange(2, tCol + 1, lr - 1, 1).getValues();
  var changed = 0, unmatched = 0;
  var block = clients.map(function (c, i) {
    var key = String(c[0]).trim();
    if (key === '') return [taxes[i][0]];
    if (!(key in taxById)) { unmatched++; return [taxes[i][0]]; }
    if (String(taxes[i][0]) !== taxById[key]) changed++;
    return [taxById[key]];
  });
  if (write && changed) {
    var range = sh.getRange(2, tCol + 1, block.length, 1);
    range.setNumberFormat('@');
    range.setValues(block);
  }
  return { tab: name, changed: changed, unmatched_client_ids: unmatched };
}

// Same keys bustTopLightCaches_ / invalidateRefsCache_ clear after a normal save.
function etRmdBustCaches_() {
  var id = ET_RMD_SS_ID, done = [];
  try { CacheService.getScriptCache().put('et_refs_ver_' + id, String(new Date().getTime()), 21600); done.push('et_refs_ver'); } catch (e) {}
  ['et_dashboard_kpis_', 'et_cust_sales_opts_', 'et_qty_map_', 'et_price_map_', 'et_cat_opts_'].forEach(function (k) {
    try { CacheService.getScriptCache().remove(k + id); removeChunkedCache_(k + id); done.push(k); } catch (e) {}
  });
  ['products', 'parties', 'categories'].forEach(function (k) {
    try { invalidateRefsCache_(id, k); done.push('refs_' + k); } catch (e) {}
  });
  ['erp_test_products', 'erp_test_customer_vendor', 'erp_test_sales_invoices', 'erp_test_sales_offer'].forEach(function (t) {
    try { noteTableChange_(id, t); done.push('stamp_' + t); } catch (e) {}
  });
  return done;
}

// ---------- generators ----------

// Small seeded PRNG (mulberry32) so every run produces the same data.
function etRmdRng_(seed) {
  var a = seed >>> 0;
  return function () {
    a = (a + 0x6D2B79F5) >>> 0;
    var t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
function etRmdShuffle_(arr, rnd) {
  for (var i = arr.length - 1; i > 0; i--) {
    var j = Math.floor(rnd() * (i + 1));
    var t = arr[i]; arr[i] = arr[j]; arr[j] = t;
  }
  return arr;
}
function etRmdPad_(n, w) { var s = String(n); while (s.length < w) s = '0' + s; return s; }
